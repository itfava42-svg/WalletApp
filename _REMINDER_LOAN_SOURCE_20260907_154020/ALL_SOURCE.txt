package com.example.walletapp.data

import android.content.Context
import androidx.compose.runtime.mutableStateListOf
import org.json.JSONArray
import org.json.JSONObject

enum class ReminderKind {
    RECEIVE,
    PAY
}

enum class RepeatFrequency(val label: String) {
    ONCE("یک بار"),
    DAILY("هر روز"),
    WEEKLY("هر هفته"),
    MONTHLY("هر ماه")
}

enum class ReminderPriority(val label: String) {
    LOW("کم"),
    MEDIUM("متوسط"),
    HIGH("زیاد"),
    URGENT("فوری")
}

enum class ReminderCategory(val label: String) {
    BILL("قبض"),
    INSTALLMENT("قسط"),
    DEBT("بدهی"),
    SALARY("حقوق"),
    PURCHASE("خرید"),
    OTHER("سایر")
}

data class Reminder(
    val id: String,
    val title: String,
    val amount: Long,
    val kind: ReminderKind,
    val description: String,
    val repeat: RepeatFrequency,
    val triggerAtMillis: Long,
    val isActive: Boolean = true,
    val priority: ReminderPriority = ReminderPriority.MEDIUM,
    val category: ReminderCategory = ReminderCategory.OTHER,
    val isCompleted: Boolean = false
)

object ReminderStore {

    val reminders = mutableStateListOf<Reminder>()

    private var context: Context? = null
    private var loaded = false

    private const val PREFS = "wallet_app_prefs"
    private const val KEY = "reminders_json"

    fun attachContext(ctx: Context) {
        if (context == null) {
            context = ctx.applicationContext
        }

        if (!loaded) {
            loaded = true
            load()
        }
    }

    private fun load() {
        val ctx = context ?: return

        val json = ctx.getSharedPreferences(
            PREFS,
            Context.MODE_PRIVATE
        ).getString(KEY, null) ?: return

        runCatching {
            val array = JSONArray(json)

            reminders.clear()

            for (i in 0 until array.length()) {
                val item = array.getJSONObject(i)

                reminders.add(
                    Reminder(
                        id = item.getString("id"),
                        title = item.getString("title"),
                        amount = item.optLong("amount", 0L),
                        kind = runCatching {
                            ReminderKind.valueOf(
                                item.optString("kind", ReminderKind.PAY.name)
                            )
                        }.getOrDefault(ReminderKind.PAY),
                        description = item.optString("description", ""),
                        repeat = runCatching {
                            RepeatFrequency.valueOf(
                                item.optString("repeat", RepeatFrequency.ONCE.name)
                            )
                        }.getOrDefault(RepeatFrequency.ONCE),
                        triggerAtMillis = item.optLong(
                            "triggerAtMillis",
                            System.currentTimeMillis()
                        ),
                        isActive = item.optBoolean("isActive", true),
                        priority = runCatching {
                            ReminderPriority.valueOf(
                                item.optString(
                                    "priority",
                                    ReminderPriority.MEDIUM.name
                                )
                            )
                        }.getOrDefault(ReminderPriority.MEDIUM),
                        category = runCatching {
                            ReminderCategory.valueOf(
                                item.optString(
                                    "category",
                                    ReminderCategory.OTHER.name
                                )
                            )
                        }.getOrDefault(ReminderCategory.OTHER),
                        isCompleted = item.optBoolean(
                            "isCompleted",
                            false
                        )
                    )
                )
            }
        }
    }

    private fun save() {
        val ctx = context ?: return

        val array = JSONArray()

        reminders.forEach {
            val item = JSONObject()

            item.put("id", it.id)
            item.put("title", it.title)
            item.put("amount", it.amount)
            item.put("kind", it.kind.name)
            item.put("description", it.description)
            item.put("repeat", it.repeat.name)
            item.put("triggerAtMillis", it.triggerAtMillis)
            item.put("isActive", it.isActive)
            item.put("priority", it.priority.name)
            item.put("category", it.category.name)
            item.put("isCompleted", it.isCompleted)

            array.put(item)
        }

        ctx.getSharedPreferences(
            PREFS,
            Context.MODE_PRIVATE
        )
            .edit()
            .putString(KEY, array.toString())
            .apply()
    }

    fun add(reminder: Reminder) {
        reminders.add(0, reminder)
        save()
    }

    fun update(reminder: Reminder) {
        val index = reminders.indexOfFirst {
            it.id == reminder.id
        }

        if (index >= 0) {
            reminders[index] = reminder
            save()
        }
    }

    fun remove(id: String) {
        reminders.removeAll {
            it.id == id
        }
        save()
    }

    fun getById(id: String): Reminder? {
        return reminders.firstOrNull {
            it.id == id
        }
    }

    fun setCompleted(
        id: String,
        completed: Boolean
    ) {
        getById(id)?.let {
            update(
                it.copy(
                    isCompleted = completed,
                    isActive = if (completed) false else it.isActive
                )
            )
        }
    }

    fun weekTotal(
        kind: ReminderKind
    ): Long {
        return reminders
            .filter {
                it.isActive &&
                    !it.isCompleted &&
                    it.kind == kind
            }
            .sumOf {
                it.amount
            }
    }

    fun monthTotal(
        kind: ReminderKind
    ): Long {
        return reminders
            .filter {
                it.isActive &&
                    !it.isCompleted &&
                    it.kind == kind
            }
            .sumOf {
                it.amount
            }
    }
}

package com.example.walletapp.loan

import android.content.Context

object LoanStore {

    private val loans = mutableListOf<Loan>()

    private var appContext: Context? = null
    private var loadedFromDisk = false

    fun attachContext(context: Context) {

        if (appContext == null) {
            appContext = context.applicationContext
        }

        if (!loadedFromDisk) {

            loadedFromDisk = true

            appContext?.let { ctx ->

                val loaded =
                    LoanPersistence.load(ctx)

                if (loaded.isNotEmpty()) {

                    loans.clear()
                    loans.addAll(loaded)

                }
            }
        }
    }

    private fun persist() {

        appContext?.let { ctx ->

            LoanPersistence.save(
                ctx,
                loans
            )
        }
    }

    fun addLoan(loan: Loan) {

        loans.add(loan)

        persist()
    }

    fun getLoans(): List<Loan> {

        return loans.toList()
    }

    fun getActiveLoans(): List<Loan> {

        return loans.filter {
            it.isActive
        }
    }

    fun findById(id: String): Loan? {

        return loans.firstOrNull {
            it.id == id
        }
    }

    fun updateLoan(loan: Loan) {

        val index =
            loans.indexOfFirst {
                it.id == loan.id
            }

        if (index >= 0) {

            loans[index] =
                loan.copy(
                    updatedAt =
                        System.currentTimeMillis()
                )

            persist()
        }
    }

    fun deleteLoan(id: String) {

        loans.removeAll {
            it.id == id
        }

        persist()
    }

    fun clear() {

        loans.clear()

        persist()
    }
}

package com.example.walletapp.reminders

import android.content.BroadcastReceiver
import android.content.Context
import android.content.Intent
import com.example.walletapp.data.ReminderStore
import com.example.walletapp.data.RepeatFrequency
import java.util.Calendar

class ReminderReceiver : BroadcastReceiver() {

    companion object {
        const val EXTRA_REMINDER_ID = "reminder_id"
    }

    override fun onReceive(
        context: Context,
        intent: Intent
    ) {
        val id = intent.getStringExtra(
            EXTRA_REMINDER_ID
        ) ?: return

        ReminderStore.attachContext(context)

        val reminder = ReminderStore.getById(id)
            ?: return

        if (!reminder.isActive || reminder.isCompleted) {
            return
        }

        ReminderNotifier.notify(
            context,
            reminder
        )

        if (reminder.repeat == RepeatFrequency.ONCE) {
            ReminderStore.update(
                reminder.copy(
                    isActive = false
                )
            )
            return
        }

        val calendar = Calendar.getInstance().apply {
            timeInMillis = reminder.triggerAtMillis

            when (reminder.repeat) {
                RepeatFrequency.DAILY ->
                    add(Calendar.DAY_OF_MONTH, 1)

                RepeatFrequency.WEEKLY ->
                    add(Calendar.DAY_OF_MONTH, 7)

                RepeatFrequency.MONTHLY -> {
                    val originalDay =
                        get(Calendar.DAY_OF_MONTH)

                    add(Calendar.MONTH, 1)

                    val maxDay =
                        getActualMaximum(Calendar.DAY_OF_MONTH)

                    set(
                        Calendar.DAY_OF_MONTH,
                        minOf(originalDay, maxDay)
                    )
                }

                RepeatFrequency.ONCE -> Unit
            }
        }

        var nextMillis = calendar.timeInMillis

        if (nextMillis <= System.currentTimeMillis()) {
            nextMillis = System.currentTimeMillis() + 60_000L
        }

        val next = reminder.copy(
            triggerAtMillis = nextMillis
        )

        ReminderStore.update(next)

        ReminderScheduler.schedule(
            context,
            next
        )
    }
}

import android.app.AlarmManager
import android.app.PendingIntent
import android.content.Context
import android.content.Intent
import android.os.Build
import com.example.walletapp.data.Reminder

object ReminderScheduler {

    private fun pendingIntent(
        context: Context,
        reminder: Reminder
    ): PendingIntent {

        val intent =
            Intent(
                context,
                ReminderReceiver::class.java
            ).apply {

                putExtra(
                    ReminderReceiver.EXTRA_REMINDER_ID,
                    reminder.id
                )
            }

        return PendingIntent.getBroadcast(
            context,
            reminder.id.hashCode(),
            intent,
            PendingIntent.FLAG_UPDATE_CURRENT or
                PendingIntent.FLAG_IMMUTABLE
        )
    }

    fun canScheduleExactAlarms(
        context: Context
    ): Boolean {

        if (Build.VERSION.SDK_INT <
            Build.VERSION_CODES.S
        ) {
            return true
        }

        val alarmManager =
            context.getSystemService(
                Context.ALARM_SERVICE
            ) as AlarmManager

        return alarmManager.canScheduleExactAlarms()
    }

    fun schedule(
        context: Context,
        reminder: Reminder
    ) {

        if (!reminder.isActive) return

        val alarmManager =
            context.getSystemService(
                Context.ALARM_SERVICE
            ) as AlarmManager

        val pending =
            pendingIntent(
                context,
                reminder
            )

        if (
            Build.VERSION.SDK_INT >=
            Build.VERSION_CODES.S &&
            alarmManager.canScheduleExactAlarms()
        ) {

            alarmManager.setExactAndAllowWhileIdle(
                AlarmManager.RTC_WAKEUP,
                reminder.triggerAtMillis,
                pending
            )

        } else {

            alarmManager.set(
                AlarmManager.RTC_WAKEUP,
                reminder.triggerAtMillis,
                pending
            )
        }
    }

    fun cancel(
        context: Context,
        reminder: Reminder
    ) {

        val alarmManager =
            context.getSystemService(
                Context.ALARM_SERVICE
            ) as AlarmManager

        alarmManager.cancel(
            pendingIntent(
                context,
                reminder
            )
        )
    }
}

@Composable
fun LoanManagementScreen(
    onBack: () -> Unit
) {
    var showEditor by remember { mutableStateOf(false) }
    var editingLoan by remember { mutableStateOf<LoanEntry?>(null) }
    var deleteLoan by remember { mutableStateOf<LoanEntry?>(null) }
    var installmentsLoan by remember { mutableStateOf<LoanEntry?>(null) }

    val totalLoans = LoanStore.loans.size

    val totalLoanAmount =
        LoanStore.loans.sumOf { it.totalAmount }

    val totalPaidAmount =
        LoanStore.loans.sumOf { loan ->
            loan.payments.sumOf { it.amount }
        }

    val totalScheduledAmount =
        LoanStore.loans.sumOf { loan ->
            loan.installmentAmount *
                loan.totalInstallments.toLong()
        }

    val totalRemainingAmount =
        (totalScheduledAmount - totalPaidAmount)
            .coerceAtLeast(0L)

    val totalPaidInstallments =
        LoanStore.loans.sumOf { loan ->
            loan.payments.size
                .coerceAtMost(loan.totalInstallments)
        }

    val totalInstallments =
        LoanStore.loans.sumOf {
            it.totalInstallments
        }

    Scaffold(
        containerColor = MatureColors.Surface,
        floatingActionButton = {
            FloatingActionButton(
                onClick = {
                    editingLoan = null
                    showEditor = true
                },
                containerColor = LoanOrange,
                contentColor = MatureColors.PrimaryDark,
                shape = RoundedCornerShape(18.dp)
            ) {
                Icon(
                    imageVector = Icons.Filled.Add,
                    contentDescription = "افزودن وام",
                    modifier = Modifier.size(27.dp)
                )
            }
        }
    ) { padding ->

        LazyColumn(
            modifier = Modifier
                .fillMaxSize()
                .padding(padding),
            contentPadding = androidx.compose.foundation.layout.PaddingValues(
                start = 18.dp,
                end = 18.dp,
                top = 16.dp,
                bottom = 100.dp
            ),
            verticalArrangement = Arrangement.spacedBy(14.dp)
        ) {

            item {
                LoanHeader(
                    totalLoans = totalLoans,
                    totalRemainingAmount = totalRemainingAmount,
                    onBack = onBack
                )
            }

            if (LoanStore.loans.isNotEmpty()) {

                item {
                    LoanSectionTitle(
                        title = "خلاصه وضعیت",
                        subtitle = "نمای کلی تعهدات مالی"
                    )
                }

                item {
                    Row(
                        modifier = Modifier.fillMaxWidth(),
                        horizontalArrangement = Arrangement.spacedBy(10.dp)
                    ) {
                        LoanSummaryTile(
                            modifier = Modifier.weight(1f),
                            title = "مجموع وام",
                            value = totalLoanAmount.toPersianCurrency(),
                            icon = Icons.Filled.Payments,
                            background = MatureColors.SecondaryLight,
                            iconTint = MatureColors.Primary
                        )

                        LoanSummaryTile(
                            modifier = Modifier.weight(1f),
                            title = "مانده بدهی",
                            value = totalRemainingAmount.toPersianCurrency(),
                            icon = Icons.Filled.AccountBalance,
                            background = MatureColors.Surface,
                            iconTint = MatureColors.PrimaryDark
                        )
                    }
                }

                item {
                    Surface(
                        modifier = Modifier.fillMaxWidth(),
                        shape = RoundedCornerShape(20.dp),
                        color = MatureColors.CardBackground
                    ) {
                        Row(
                            modifier = Modifier
                                .fillMaxWidth()
                                .padding(13.dp),
                            horizontalArrangement = Arrangement.spacedBy(9.dp)
                        ) {
                            LoanMiniSummary(
                                modifier = Modifier.weight(1f),
                                title = "کل اقساط",
                                value = totalInstallments.toString()
                            )

                            LoanMiniSummary(
                                modifier = Modifier.weight(1f),
                                title = "پرداخت‌شده",
                                value = totalPaidInstallments.toString()
                            )

                            LoanMiniSummary(
                                modifier = Modifier.weight(1f),
                                title = "باقی‌مانده",
                                value = (
                                    totalInstallments -
                                        totalPaidInstallments
                                    ).coerceAtLeast(0).toString()
                            )
                        }
                    }
                }

                item {
                    LoanSectionTitle(
                        title = "فهرست وام‌ها",
                        subtitle = "برای مشاهده جزئیات هر وام انتخاب کنید"
                    )
                }

                items(
                    items = LoanStore.loans,
                    key = { it.id }
                ) { loan ->
                    LoanCard(
                        loan = loan,
                        onEdit = {
                            editingLoan = loan
                            showEditor = true
                        },
                        onDelete = {
                            deleteLoan = loan
                        },
                        onInstallments = {
                            installmentsLoan = loan
                        }
                    )
                }

            } else {

                item {
                    EmptyLoanState(
                        onAdd = {
                            editingLoan = null
                            showEditor = true
                        }
                    )
                }
            }
        }
    }

    if (showEditor) {
        LoanEditorDialog(
            existingLoan = editingLoan,
            onDismiss = {
                showEditor = false
                editingLoan = null
            },
            onSave = { loan ->
                if (editingLoan == null) {
                    LoanStore.add(loan)
                } else {
                    LoanStore.update(loan)
                }

                showEditor = false
                editingLoan = null
            }
        )
    }

    installmentsLoan?.let { loan ->
        LoanInstallmentsDialog(
            loan = loan,
            onDismiss = {
                installmentsLoan = null
            }
        )
    }

    deleteLoan?.let { loan ->
        AlertDialog(
            onDismissRequest = {
                deleteLoan = null
            },
            title = {
                Text(
                    text = "حذف وام",
                    fontWeight = FontWeight.Bold,
                    color = MatureColors.TextPrimary
                )
            },
            text = {
                Text(
                    text = "آیا از حذف وام «${loan.title}» مطمئن هستید؟",
                    color = MatureColors.TextSecondary
                )
            },
            confirmButton = {
                Button(
                    onClick = {
                        LoanStore.remove(loan.id)
                        deleteLoan = null
                    },
                    colors = ButtonDefaults.buttonColors(
                        containerColor = MatureColors.PrimaryDark
                    ),
                    shape = RoundedCornerShape(12.dp)
                ) {
                    Text("حذف")
                }
            },
            dismissButton = {
                OutlinedButton(
                    onClick = {
                        deleteLoan = null
                    },
                    shape = RoundedCornerShape(12.dp)
                ) {
                    Text("انصراف")
                }
            }
        )
    }
}

@Composable
private fun LoanCard(
    loan: LoanEntry,
    onEdit: () -> Unit,
    onDelete: () -> Unit,
    onInstallments: () -> Unit
) {
    val paidCount =
        loan.payments.size.coerceAtMost(loan.totalInstallments)

    val remainingCount =
        (loan.totalInstallments - paidCount)
            .coerceAtLeast(0)

    val paidAmount =
        loan.payments.sumOf { it.amount }

    val totalScheduledAmount =
        loan.installmentAmount *
            loan.totalInstallments.toLong()

    val remainingAmount =
        (totalScheduledAmount - paidAmount)
            .coerceAtLeast(0L)

    val progress =
        if (loan.totalInstallments > 0) {
            paidCount.toFloat() /
                loan.totalInstallments.toFloat()
        } else {
            0f
        }

    Card(
        modifier = Modifier.fillMaxWidth(),
        shape = RoundedCornerShape(24.dp),
        colors = CardDefaults.cardColors(
            containerColor = MatureColors.CardBackground
        ),
        elevation = CardDefaults.cardElevation(
            defaultElevation = 3.dp
        )
    ) {
        Column(
            modifier = Modifier.padding(16.dp),
            verticalArrangement = Arrangement.spacedBy(13.dp)
        ) {

            Row(
                modifier = Modifier.fillMaxWidth(),
                verticalAlignment = Alignment.CenterVertically
            ) {
                Surface(
                    modifier = Modifier.size(48.dp),
                    shape = RoundedCornerShape(15.dp),
                    color = MatureColors.SecondaryLight
                ) {
                    Box(
                        contentAlignment = Alignment.Center
                    ) {
                        Icon(
                            imageVector = Icons.Filled.AccountBalance,
                            contentDescription = null,
                            tint = MatureColors.Primary,
                            modifier = Modifier.size(25.dp)
                        )
                    }
                }

                Spacer(Modifier.size(11.dp))

                Column(
                    modifier = Modifier.weight(1f),
                    verticalArrangement = Arrangement.spacedBy(3.dp)
                ) {
                    Text(
                        text = loan.title,
                        fontSize = 17.sp,
                        fontWeight = FontWeight.Bold,
                        color = MatureColors.TextPrimary
                    )

                    Text(
                        text = loan.type.label,
                        fontSize = 12.sp,
                        color = MatureColors.TextSecondary
                    )
                }

                IconButton(onClick = onEdit) {
                    Icon(
                        imageVector = Icons.Filled.Edit,
                        contentDescription = "ویرایش",
                        tint = MatureColors.Primary
                    )
                }

                IconButton(onClick = onDelete) {
                    Icon(
                        imageVector = Icons.Filled.Delete,
                        contentDescription = "حذف",
                        tint = MatureColors.PrimaryDark
                    )
                }
            }

            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.spacedBy(9.dp)
            ) {
                LoanInfoTile(
                    modifier = Modifier.weight(1f),
                    title = "مبلغ وام",
                    value = loan.totalAmount.toPersianCurrency(),
                    background = MatureColors.SecondaryLight,
                    icon = Icons.Filled.Payments,
                    iconTint = MatureColors.Primary
                )

                LoanInfoTile(
                    modifier = Modifier.weight(1f),
                    title = "مبلغ هر قسط",
                    value = loan.installmentAmount.toPersianCurrency(),
                    background = MatureColors.Surface,
                    icon = Icons.Filled.Schedule,
                    iconTint = MatureColors.Success
                )
            }

            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.spacedBy(8.dp)
            ) {
                LoanDetailChip(
                    modifier = Modifier.weight(1f),
                    title = "نرخ سود",
                    value = "${loan.interestRatePercent}٪",
                    icon = Icons.Filled.Percent
                )

                LoanDetailChip(
                    modifier = Modifier.weight(1f),
                    title = "کل اقساط",
                    value = loan.totalInstallments.toString(),
                    icon = Icons.Filled.Schedule
                )

                LoanDetailChip(
                    modifier = Modifier.weight(1f),
                    title = "باقی",
                    value = remainingCount.toString(),
                    icon = Icons.Filled.CalendarMonth
                )
            }

            Column(
                verticalArrangement = Arrangement.spacedBy(7.dp)
            ) {
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Text(
                        text = "پیشرفت بازپرداخت",
                        fontSize = 12.sp,
                        color = MatureColors.TextSecondary
                    )

                    Spacer(Modifier.weight(1f))

                    Text(
                        text = "$paidCount از ${loan.totalInstallments} قسط",
                        fontSize = 12.sp,
                        fontWeight = FontWeight.Bold,
                        color = MatureColors.Primary
                    )
                }

                LinearProgressIndicator(
                    progress = { progress.coerceIn(0f, 1f) },
                    modifier = Modifier
                        .fillMaxWidth()
                        .height(8.dp)
                        .clip(RoundedCornerShape(10.dp)),
                    color = MatureColors.Success,
                    trackColor = MatureColors.SecondaryLight
                )
            }

            Surface(
                modifier = Modifier.fillMaxWidth(),
                shape = RoundedCornerShape(17.dp),
                color = if (remainingCount == 0) {
                    MatureColors.Surface
                } else {
                    MatureColors.SecondaryLight.copy(alpha = 0.70f)
                }
            ) {
                Row(
                    modifier = Modifier
                        .fillMaxWidth()
                        .padding(
                            horizontal = 13.dp,
                            vertical = 11.dp
                        ),
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    Column(
                        modifier = Modifier.weight(1f)
                    ) {
                        Text(
                            text = if (remainingCount == 0) {
                                "وضعیت وام"
                            } else {
                                "مانده بدهی"
                            },

private fun LoanInstallmentsDialog(

private fun calculateLoanInstallmentDate(
    year: Int,
    month: Int,
    day: Int,
    monthsToAdd: Int
): String {
    val totalMonths =
        (year * 12) +
            (month - 1) +
            monthsToAdd

    val newYear =
        totalMonths / 12

    val newMonth =
        (totalMonths % 12) + 1

    val maxDay =
        if (newMonth <= 6) {
            31
        } else {
            30
        }

    val newDay =
        day.coerceAtMost(maxDay)

    return "$newYear/${newMonth.toString().padStart(2, '0')}/${newDay.toString().padStart(2, '0')}"
}

    var amountText by remember {
        mutableStateOf(
            existingLoan?.totalAmount?.toString() ?: ""
        )
    }

        mutableStateOf(
            existingLoan?.interestRatePercent?.toString() ?: ""
        )

    }

        mutableStateOf(
            existingLoan?.totalInstallments?.toString() ?: ""

    }

        mutableStateOf(
            existingLoan?.intervalMonths ?: 1
        )
    }

        mutableStateOf(
            existingLoan?.let {
                "${it.startYear}/${it.startMonth.toString().padStart(2, '0')}/${it.startDay.toString().padStart(2, '0')}"

        )

            existingLoan?.let {

                    "${it.firstInstallmentYear}/${it.firstInstallmentMonth.toString().padStart(2, '0')}/${it.firstInstallmentDay.toString().padStart(2, '0')}"

                            title = it

                onClick = onDismiss,
                shape = RoundedCornerShape(14.dp)

            }
        }

private fun parseJalaliDate(
    value: String
): IntArray? {
    val parts = value.split("/")

    if (parts.size != 3) {
        return null
    }

    val year =
        parts[0].toIntOrNull()
            ?: return null

    val month =
        parts[1].toIntOrNull()
            ?: return null

    val day =
        parts[2].toIntOrNull()
            ?: return null

    if (
        month !in 1..12 ||
        day !in 1..31
    ) {
        return null
    }

    return intArrayOf(
        year,
        month,
        day
    )
}

private fun calculateMonthlyInstallment(
    principal: Long,
    annualInterestPercent: Double,
    numberOfPayments: Int
): Long {
    if (
        principal <= 0L ||
        numberOfPayments <= 0
    ) {
        return 0L
    }

    if (annualInterestPercent <= 0.0) {
        return principal / numberOfPayments
    }

    val monthlyRate =
        annualInterestPercent /
            100.0 /
            12.0

    val power =
        (1.0 + monthlyRate)
            .pow(numberOfPayments)

    val payment =
        principal *
            (monthlyRate * power) /
            (power - 1.0)

    return payment.toLong()
}
